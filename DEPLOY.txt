═══════════════════════════════════════════════════════════════════
 PASO 4 — SET JWT_SECRET EN PRODUCCIÓN (CRÍTICO — V1 fix)
═══════════════════════════════════════════════════════════════════

  # ⚠️  Sin esto, el worker rechazará todos los logins en producción
  #     (V1 fix exige JWT_SECRET en env.ENVIRONMENT === 'production')

  # Generar un secret aleatorio:
  openssl rand -base64 48

  # Ponerlo como secret en Cloudflare:
  npx wrangler secret put JWT_SECRET
  # (pegar el valor generado arriba y pulsar Enter)

  # Configurar ENVIRONMENT=production:
  # Editar wrangler.toml y añadir en [vars]:
  #   ENVIRONMENT = "production"
  # O usar:
  npx wrangler secret put ENVIRONMENT
  # (escribir "production" y pulsar Enter)

═══════════════════════════════════════════════════════════════════
 PASO 8 — VERIFICACIÓN DE SEGURIDAD (S113-W)
═══════════════════════════════════════════════════════════════════

  # V2: Queue cross-user protection
  # Intentar leer la cola de otro usuario sin autenticación:
  curl -i https://el-bunker-api.thebunker-admin.workers.dev/api/queue/edu
  # Debe devolver: 401 Unauthorized

  # V11: Rate limiting
  # Lanzar 15 requests rápidos:
  for i in $(seq 1 15); do
    curl -s -o /dev/null -w "%{http_code}\n" \
      https://el-bunker-api.thebunker-admin.workers.dev/api/health
  done
  # Los últimos ~5 deben devolver: 429 Too Many Requests

  # V3/V4: Path traversal y upload prefix validation
  curl -X POST https://el-bunker-api.thebunker-admin.workers.dev/api/upload \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer TOKEN_VALIDO" \
    -d '{"prefix":"malicioso","filename":"../../etc/passwd"}'
  # Debe devolver: 400 Invalid prefix
