const _loadGapiScript = () =>
  new Promise((resolve, reject) => {
    if (window.gapi) return resolve();
    const script   = document.createElement("script");
    script.src     = "https://apis.google.com/js/api.js";
    script.onload  = resolve;
    script.onerror = () => reject(_vaultError("GAPI_LOAD", "No se pudo cargar gapi.js"));
    document.head.appendChild(script);
  });

const _loadGisScript = () =>
  new Promise((resolve, reject) => {
    if (window.google?.accounts) return resolve();
    const script   = document.createElement("script");
    script.src     = "https://accounts.google.com/gsi/client";
    script.onload  = resolve;
    script.onerror = () => reject(_vaultError("GIS_LOAD", "No se pudo cargar GIS client"));
    document.head.appendChild(script);
  });

const _initGapiClient = () =>
  new Promise((resolve, reject) => {
    window.gapi.load("client", {
      callback: async () => {
        try {
          await window.gapi.client.init({});
          await window.gapi.client.load(
            "https://www.googleapis.com/discovery/v1/apis/drive/v3/rest"
          );
          _state.gapiReady = true;
          resolve();
        } catch (err) {
          reject(_vaultError("GAPI_INIT", "Error inicializando gapi.client", err));
        }
      },
      onerror: (err) =>
        reject(_vaultError("GAPI_CLIENT_LOAD", "Error cargando gapi.client", err)),
    });
  });

/**
 * requestAccessToken(forceConsent)
 * Solicita (o renueva) el access token OAuth2 vÃ­a GIS.
 */
export const requestAccessToken = (forceConsent = false) =>
  new Promise((resolve, reject) => {
    if (!_state.tokenClient) {
      return reject(_vaultError("NOT_INIT", "Llama a initGoogleServices() primero"));
    }

    _state.tokenClient.callback = async (tokenResponse) => {
      if (tokenResponse.error) {
        _notifyAuthListeners("AUTH_ERROR", { error: tokenResponse });
        return reject(
          _vaultError("TOKEN_ERROR", tokenResponse.error_description, tokenResponse)
        );
      }

      _state.accessToken = tokenResponse.access_token;
      window.gapi.client.setToken({ access_token: tokenResponse.access_token });
      const expiresAt = Date.now() + (tokenResponse.expires_in ?? 3600) * 1000;
      localStorage.setItem("bunker_token", tokenResponse.access_token);
      localStorage.setItem("bunker_token_exp", String(expiresAt));
      _notifyAuthListeners("SIGNED_IN", { token: tokenResponse.access_token });
      console.info("[BÃºnker] Token OAuth2 obtenido âœ“");

      try {
        await _triggerHeartbeat();
      } catch (hbErr) {
        console.warn("[BÃºnker] Heartbeat post-login fallido:", hbErr);
      }

      resolve(tokenResponse.access_token);
    };

    if (forceConsent) {
      _state.tokenClient.requestAccessToken({ prompt: "select_account consent" });
    } else {
      _state.tokenClient.requestAccessToken({ prompt: "" });
    }
  });

/** signOut() â€” Revoca el token y limpia el estado local. */
export const signOut = () => {
  if (_state.accessToken) {
    window.google.accounts.oauth2.revoke(_state.accessToken, () => {
      console.info("[BÃºnker] Token revocado");
    });
  }
  _state.accessToken = null;
  localStorage.removeItem("bunker_token");
  localStorage.removeItem("bunker_token_exp");
  _state.dbFileId    = null;
  window.gapi.client.setToken(null);
  _notifyAuthListeners("SIGNED_OUT");
};

/** @returns {boolean} */
export const restoreToken = () => {
  try {
    const token = localStorage.getItem("bunker_token");
    const exp   = Number(localStorage.getItem("bunker_token_exp") ?? 0);
    if (!token || Date.now() > exp) {
      localStorage.removeItem("bunker_token");
      localStorage.removeItem("bunker_token_exp");
      return false;
    }
    _state.accessToken = token;
    window.gapi.client.setToken({ access_token: token });
    console.info("[BÃºnker] Token restaurado desde localStorage âœ“");
    return true;
  } catch (err) {
    console.warn("[BÃºnker] No se pudo restaurar el token:", err);
    return false;
  }
};

export const isAuthenticated = () => Boolean(_state.accessToken);

/**
 * onAuthChange(callback)
 * SuscripciÃ³n a eventos: SIGNED_IN | SIGNED_OUT | AUTH_ERROR
 * @returns {Function} unsubscribe
 */
export const onAuthChange = (callback) => {
  _state.authListeners.push(callback);
  return () => {
    _state.authListeners = _state.authListeners.filter((fn) => fn !== callback);
  };
};

const _findDatabaseFile = async () => {
  const folderId = import.meta.env.VITE_GOOGLE_SHARED_FOLDER_ID;
  if (!folderId) {
    throw _vaultError("MISSING_FOLDER_ID", "VITE_GOOGLE_SHARED_FOLDER_ID no estÃ¡ configurada. Revisa el .env.");
  }
  if (_state.dbFileId) return _state.dbFileId;

  console.info(`[BÃºnker] Buscando DB en carpeta compartida: ${folderId}`);

  try {
    const params = {
      spaces: "drive",
      q:      `'${folderId}' in parents and name = '${CONFIG.DB_FILENAME}' and trashed = false`,
      fields: "files(id, name, modifiedTime)",
      includeItemsFromAllDrives: true,
      supportsAllDrives:         true,
    };
    const response = await window.gapi.client.drive.files.list(params);

    const files = response.result.files ?? [];
    if (files.length === 0) return null;

    if (files.length > 1) {
      console.warn("[BÃºnker] MÃºltiples archivos DB â€” usando el mÃ¡s reciente");
      files.sort((a, b) => new Date(b.modifiedTime) - new Date(a.modifiedTime));
    }

    _state.dbFileId = files[0].id;
    return _state.dbFileId;
  } catch (err) {
    throw _vaultError("FIND_FILE", "Error buscando el archivo DB en Drive", err);
  }
};

const _createDatabaseFile = async (data) => {

    const body          = JSON.stringify(data, null, 2);

const _withTokenRetry = async (fn) => {
  try {
    return await fn();
  } catch (err) {
    const status = err?.status ?? err?.result?.error?.code ?? err?.code;
    if (status === 401 || status === 403) {
      console.info("[BÃºnker] Token rechazado (HTTP " + status + ") â€” renovandoâ€¦");
      try {
        await requestAccessToken(false);
      } catch (_silentErr) {
        console.warn("[BÃºnker] Silent refresh fallido â€” forzando popup de loginâ€¦");
        await requestAccessToken(true);
      }
      return await fn();
    }
    throw err;
  }
};

      path:    `https://www.googleapis.com/upload/drive/v3/files/${fileId}`,
      method:  "PATCH",
      params:  { uploadType: "multipart", supportsAllDrives: true },
      headers: { "Content-Type": `multipart/related; boundary="${boundary}"` },
      body:    multipartBody,
    });
  } catch (err) {
    throw _vaultError("UPDATE_QUEUE", `Error actualizando cola de ${userId} en Drive`, err);
  }
};

/**
 * initDatabase()
 * Inicializa la DB: la crea si no existe, la migra si hay versiÃ³n antigua.
 */
export const initDatabase = async () => {
  if (!isAuthenticated()) {
    throw _vaultError("NOT_AUTH", "Usuario no autenticado. Llama a requestAccessToken() primero.");
  }

  try {
    const fileId = await _findDatabaseFile();

    if (!fileId) {
      console.info("[BÃºnker] Primera ejecuciÃ³n â€” creando DBâ€¦");
      const freshDb = EMPTY_DB();
      await _createDatabaseFile(freshDb);
      return freshDb;
    }

    const existing = await _readDatabaseFile(fileId);
    const migrated = _migrateSchema(existing);

    if (migrated._wasMigrated) {
      delete migrated._wasMigrated;
      await _updateDatabaseFile(migrated, fileId);
      console.info("[BÃºnker] Esquema migrado y guardado âœ“");
    }

    console.info("[BÃºnker] DB cargada desde Drive âœ“");
    return migrated;
  } catch (err) {
    if (err.code) throw err;
    throw _vaultError("INIT_DB", "Error inicializando la base de datos", err);
  }
};

/**
 * updateData(updater)
 * PatrÃ³n read-modify-write seguro.
 * @param {Function} updater â€” fn(currentDb) => newDb
 */
export const updateData = async (updater) => {
  if (!isAuthenticated()) {
    throw _vaultError("NOT_AUTH", "Usuario no autenticado.");
  }
  if (typeof updater !== "function") {
    throw _vaultError("INVALID_UPDATER", "updateData() requiere una funciÃ³n como argumento.");
  }

  try {
    const fileId  = await _findDatabaseFile();
    if (!fileId) {
      throw _vaultError("NO_DB", "DB no encontrada. Llama a initDatabase() primero.");
    }

    const current = await _readDatabaseFile(fileId);
    const updated = updater(current);

    if (!updated || typeof updated !== "object") {
      throw _vaultError("INVALID_UPDATE", "La funciÃ³n updater debe retornar un objeto vÃ¡lido.");
    }

    updated.updated_at = new Date().toISOString();

    await _updateDatabaseFile(updated, fileId);
    console.info("[BÃºnker] DB actualizada en Drive âœ“");
    return updated;
  } catch (err) {
    if (err.code) throw err;
    throw _vaultError("UPDATE_DATA", "Error actualizando la base de datos", err);
  }
};

const _triggerHeartbeat = async () => {
  const now = Date.now();

  if (now - _state.lastHeartbeat < CONFIG.HEARTBEAT_THROTTLE_MS) {
    console.info("[BÃºnker] Heartbeat throttled â€” aÃºn no es necesario");
    return;
  }

  try {
    const fileId = await _findDatabaseFile();
    if (!fileId) {
      console.info("[BÃºnker] Heartbeat omitido â€” DB aÃºn no creada");
      return;
    }

    const current = await _readDatabaseFile(fileId);
    const updated = {
      ...current,
      heartbeat:  new Date().toISOString(),
      updated_at: new Date().toISOString(),
    };

    await _updateDatabaseFile(updated, fileId);
    _state.lastHeartbeat = now;

    console.info(`[BÃºnker] ðŸ’š Heartbeat registrado: ${updated.heartbeat}`);
  } catch (err) {
    console.warn("[BÃºnker] Heartbeat fallÃ³ silenciosamente:", err);
  }
};

const _migrateSchema = (data) => {
  const version = data.schema_version ?? "0.0.0";
  if (version === CONFIG.SCHEMA_VERSION) return data;

  console.info(`[BÃºnker] Migrando esquema ${version} â†’ ${CONFIG.SCHEMA_VERSION}`);

  let migrated = { ...data };

  // â”€â”€ 0.0.0 â†’ 1.0.0: campos base â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  if (!migrated.schema_version) {
    migrated = {
      ...EMPTY_DB(),
      ...migrated,
      schema_version: "1.0.0",
      music_lab: {
        tracks: migrated.music_lab?.tracks ?? [],
        drops:  migrated.music_lab?.drops  ?? [],
      },
    };
  }

  // â”€â”€ 1.0.0 â†’ 2.0.0: campo libraries â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  if (migrated.schema_version === "1.0.0") {
    migrated = {
      ...migrated,
      libraries:      migrated.libraries ?? [],
      schema_version: "2.0.0",
    };
  }

  // â”€â”€ 2.0.0 â†’ 3.0.0: campos from/to en entries â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  // AÃ±ade from:"admin" y to:"both" en entradas existentes que no lo tengan.
  if (migrated.schema_version === "2.0.0") {
    migrated = {
      ...migrated,
      entries: (migrated.entries ?? []).map((e) => ({
        ...e,
        from: e.from ?? "admin",
        to:   e.to   ?? "both",
      })),
      schema_version: "3.0.0",
    };
  }

  // â”€â”€ 3.0.0 â†’ 3.1.0: campos vibe, thumbnail, play_count, last_played en tracks
  if (migrated.schema_version === "3.0.0") {
    migrated = {
      ...migrated,
      music_lab: {
        ...migrated.music_lab,
        tracks: (migrated.music_lab?.tracks ?? []).map((t) => ({
          ...t,
          vibe:        t.vibe        ?? null,
          thumbnail:   t.thumbnail   ?? null,
          play_count:  t.play_count  ?? 0,
          last_played: t.last_played ?? null,
        })),
      },
      schema_version: "3.1.0",
    };
  }
  if (migrated.schema_version === "3.1.0") {
    migrated = {
      ...migrated,
      music_lab: {
        ...migrated.music_lab,
        playlists: migrated.music_lab?.playlists ?? [],
      },
      schema_version: "3.2.0",
    };
  }
  // â”€â”€ 3.2.0 â†’ 3.3.0: campos type y drive_file_id en tracks â”€â”€â”€â”€â”€
  if (migrated.schema_version === "3.2.0") {
    migrated = {
      ...migrated,
      music_lab: {
        ...migrated.music_lab,
        tracks: (migrated.music_lab?.tracks ?? []).map((t) => ({
          ...t,
          type:          t.type          ?? "youtube",
          drive_file_id: t.drive_file_id ?? null,
          youtube_url:   t.youtube_url   ?? t.url ?? null,
        })),
      },
      schema_version: "3.3.0",
    };
  }
    // â”€â”€ 3.3.0 â†’ 3.4.0: campos favorited y favorited_by en drops â”€â”€
  if (migrated.schema_version === "3.3.0") {
    migrated = {
      ...migrated,
      music_lab: {
        ...migrated.music_lab,
        drops: (migrated.music_lab?.drops ?? []).map((d) => ({
          ...d,
          from:         d.from         ?? null,
          to:           d.to           ?? null,
          favorited:    d.favorited    ?? false,
          favorited_by: d.favorited_by ?? null,
        })),
      },
      schema_version: "3.4.0",
    };
  }

  migrated.schema_version = CONFIG.SCHEMA_VERSION;
  migrated._wasMigrated   = true;
  return migrated;
};

/**
 * _calcLibraryStatus(lastHeartbeatCheck)
 *
 * Calcula el estado de una biblioteca segÃºn los meses
 * transcurridos desde su Ãºltimo heartbeat manual.
 * Solo aritmÃ©tica de fechas â€” no hace ninguna llamada de red.
 *
 * @param {string | null} lastHeartbeatCheck â€” ISO string o null
 * @returns {"active" | "at_risk" | "lost"}
 */
export const _calcLibraryStatus = (lastHeartbeatCheck) => {
  if (!lastHeartbeatCheck) return "at_risk"; // nunca registrado â†’ alerta inmediata

  const diffMs     = Date.now() - new Date(lastHeartbeatCheck).getTime();
  const diffMonths = diffMs / (1000 * 60 * 60 * 24 * 30.44);

  if (diffMonths >= CONFIG.LIBRARY_LOST_MONTHS)    return "lost";
  if (diffMonths >= CONFIG.LIBRARY_AT_RISK_MONTHS)  return "at_risk";
  return "active";
};

/**
 * listLibraryContents(sharedFolderId)
 *
 * Lista todos los archivos de una carpeta compartida usando el
 * token de la cuenta principal (sin login adicional).
 * Pagina automÃ¡ticamente hasta obtener todos los resultados.
 *
 * @param {string} sharedFolderId â€” ID de la carpeta compartida
 * @returns {Promise<Array>} â€” lista de archivos { id, name, size, mimeType, createdTime }
 */
export const listLibraryContents = async (sharedFolderId) => {
  if (!isAuthenticated()) {
    throw _vaultError("NOT_AUTH", "Usuario no autenticado.");
  }

  const allFiles   = [];
  let   pageToken  = null;

  try {
    do {
      const params = {
        q:                         `'${sharedFolderId}' in parents and trashed = false`,
        fields:                    "nextPageToken, files(id, name, size, mimeType, createdTime)",
        // Imprescindible para acceder a carpetas de otras cuentas compartidas
        includeItemsFromAllDrives: true,
        supportsAllDrives:         true,
        pageSize:                  1000,
      };
      if (pageToken) params.pageToken = pageToken;

      const response = await window.gapi.client.drive.files.list(params);
      const result   = response.result;

      allFiles.push(...(result.files ?? []));
      pageToken = result.nextPageToken ?? null;
    } while (pageToken);

    return allFiles;
  } catch (err) {
    throw _vaultError(
      "LIST_LIBRARY",
      `Error listando la carpeta compartida ${sharedFolderId}`,
      err
    );
  }
};

/**
 * calculateLibraryUsage(sharedFolderId)
 *
 * Suma el tamaÃ±o de todos los archivos de la carpeta compartida.
 *
 * @param {string} sharedFolderId
 * @returns {Promise<{ used_bytes: number, used_gb: number, file_count: number }>}
 */
export const calculateLibraryUsage = async (sharedFolderId) => {
  const files      = await listLibraryContents(sharedFolderId);
  const used_bytes = files.reduce((acc, f) => acc + Number(f.size ?? 0), 0);

  return {
    used_bytes,
    used_gb:    parseFloat((used_bytes / (1024 ** 3)).toFixed(2)),
    file_count: files.length,
  };
};

/**
 * checkLibraryCapacity(library)
 *
 * Comprueba si una biblioteca supera el umbral de capacidad.
 * Retorna el porcentaje usado y si se debe mostrar alerta.
 *
 * @param {{ shared_folder_id, capacity_gb }} library
 * @returns {Promise<{ used_gb, used_pct, near_full, file_count }>}
 */
export const checkLibraryCapacity = async (library) => {
  const { used_gb, file_count } = await calculateLibraryUsage(library.shared_folder_id);
  const capacity_gb = library.capacity_gb ?? 15;
  const used_pct    = parseFloat(((used_gb / capacity_gb) * 100).toFixed(1));
  const near_full   = used_pct >= CONFIG.LIBRARY_ALERT_THRESHOLD_PCT;

  return { used_gb, used_pct, near_full, file_count };
};

/**
 * fetchQueue(userId)
 * Descarga la cola del usuario desde Drive.
 * Si no existe el archivo, devuelve [] sin error.
 *
 * @param {string} userId â€” "edu" | "david" | "admin"
 * @returns {Promise<string[]>} â€” array de track_ids
 */
export const fetchQueue = async (userId) => {
  if (!isAuthenticated()) {
    throw _vaultError("NOT_AUTH", "Usuario no autenticado.");
  }
  if (!userId) {
    throw _vaultError("INVALID_USER", "fetchQueue() requiere un userId.");
  }

  try {
    const fileId = await _findQueueFile(userId);
    if (!fileId) return [];

    const response = await window.gapi.client.drive.files.get({
      fileId: fileId,
      alt:    "media",
    });

    const raw  = typeof response.body === "string"
      ? response.body
      : JSON.stringify(response.result);
    const data = JSON.parse(raw);
    return Array.isArray(data.track_ids) ? data.track_ids : [];
  } catch (err) {
    if (err.code) throw err;
    throw _vaultError("FETCH_QUEUE", `Error obteniendo cola de ${userId} desde Drive`, err);
  }
};

/**

 *
 * @param {string}   userId   â€” "edu" | "david" | "admin"
 * @param {string[]} trackIds â€” array ordenado de track_ids

export const saveQueue = async (userId, trackIds) => {
  if (!isAuthenticated()) {
    throw _vaultError("NOT_AUTH", "Usuario no autenticado.");
  }
  if (!userId) {
    throw _vaultError("INVALID_USER", "saveQueue() requiere un userId.");
  }

  const data = {
    user_id:    userId,
    track_ids:  Array.isArray(trackIds) ? trackIds : [],
    updated_at: new Date().toISOString(),
  };

  try {
    let fileId = await _findQueueFile(userId);
    if (!fileId) {
      await _createQueueFile(userId, data);
    } else {
      await _updateQueueFile(userId, data, fileId);
    }
    console.info(`[BÃºnker] Cola de ${userId} guardada âœ“ (${data.track_ids.length} tracks)`);
  } catch (err) {
    if (err.code) throw err;
    throw _vaultError("SAVE_QUEUE", `Error guardando cola de ${userId} en Drive`, err);
  }
};

/**
 * addTrackToPlaylist(playlistId, trackId)
 * AÃ±ade un track a una playlist. Idempotente: no duplica si ya existe.
 * @param {string} playlistId
 * @param {string} trackId
 * @returns {Promise<object>} â€” la playlist actualizada
 */
export const addTrackToPlaylist = async (playlistId, trackId) => {
  let updated = null;

  await updateData((db) => ({
    ...db,
    music_lab: {
      ...db.music_lab,
      playlists: (db.music_lab?.playlists ?? []).map((pl) => {
        if (pl.id !== playlistId) return pl;
        if (pl.track_ids.includes(trackId)) return pl;
        updated = {
          ...pl,
          track_ids:  [...pl.track_ids, trackId],
          updated_at: new Date().toISOString(),
        };
        return updated;
      }),
    },
  }));

  if (!updated) throw _vaultError("PLAYLIST_NOT_FOUND", `Playlist ${playlistId} no encontrada`);
  return updated;
};

/**
 * deletePlaylist(playlistId)
 * Elimina una playlist por id.
 * @param {string} playlistId
 * @returns {Promise<void>}
 */
export const deletePlaylist = async (playlistId) => {
  const db = await fetchData();

    throw _vaultError("PLAYLIST_NOT_FOUND", `Playlist ${playlistId} no encontrada`);
  }

  await updateData((db) => ({
    ...db,
    music_lab: {
      ...db.music_lab,
      playlists: (db.music_lab?.playlists ?? []).filter((pl) => pl.id !== playlistId),
    },
  }));
};

export const uploadMusicFile = async (file) => {
  const folderId = import.meta.env.VITE_GOOGLE_SHARED_FOLDER_ID;
  if (!folderId) throw _vaultError("MISSING_FOLDER_ID", "VITE_GOOGLE_SHARED_FOLDER_ID no configurada.");

  const token = window.gapi.client.getToken()?.access_token;
  if (!token) throw _vaultError("NO_TOKEN", "No hay token OAuth2 activo.");

  const metadata = {
    name:    file.name,
    mimeType: file.type || "audio/mpeg",
    parents: [folderId],
  };

  const form = new FormData();
  form.append("metadata", new Blob([JSON.stringify(metadata)], { type: "application/json" }));
  form.append("file", file);

  const response = await fetch(
    "https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id",
    {
      method:  "POST",
      headers: { Authorization: `Bearer ${token}` },
      body:    form,
    }
  );

  if (!response.ok) {
    const err = await response.json().catch(() => ({}));
    throw _vaultError("UPLOAD_MUSIC", `Error subiendo archivo a Drive: ${response.status}`, err);
  }

  const result = await response.json();
  console.info(`[BÃºnker] Archivo de mÃºsica subido âœ“ (fileId: ${result.id})`);
  return result.id;
};

export function PlayerProvider({ children, userId, tracks = [] }) {
  const [activeTrack,    setActiveTrack]    = useState(tracks[0] ?? null);
  const [isPlaying,      setIsPlaying]      = useState(false);
  const [localQueue,     setLocalQueue]     = useState([]);
  const [loopMode,       setLoopMode]       = useState("off");
  const [shuffle,        setShuffle]        = useState(false);
  const [autoPlaySignal, setAutoPlaySignal] = useState(0);
  const [syncStatus,     setSyncStatus]     = useState("idle");
  const [activeVibe,     setActiveVibe]     = useState("all");
  const [masterVolume,   setMasterVolume]   = useState(80);

  const audioRef    = useRef(null);
  const ytPlayerRef = useRef(null);
  const wakeLockRef = useRef(null);

  const isYouTube = activeTrack?.type === "youtube" || (!activeTrack?.type && activeTrack?.url);

  // â”€â”€ Persistencia cola en Drive â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    if (!userId) return;
    fetchQueue(userId)
      .then((trackIds) => {
        if (!trackIds.length) return;
        const qt = trackIds.map((id) => tracks.find((t) => t.id === id)).filter(Boolean);
        if (!qt.length) return;
        setLocalQueue(qt);
        setActiveTrack(qt[0]);
      })
      .catch((err) => console.warn("[BÃºnker] fetchQueue:", err));
  }, [userId]);

  useEffect(() => {
    if (!userId) return;
    const ids = activeTrack
      ? [activeTrack.id, ...localQueue.filter((t) => t.id !== activeTrack.id).map((t) => t.id)]
      : localQueue.map((t) => t.id);
    if (!ids.length) return;
    setSyncStatus("syncing");
    saveQueue(userId, ids)
      .then(() => setSyncStatus("idle"))
      .catch(() => { setSyncStatus("error"); setTimeout(() => setSyncStatus("idle"), 4000); });
  }, [activeTrack?.id, localQueue, userId]);

  // â”€â”€ Skip â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const skip = useCallback((dir, filteredTracks) => {
    // Cola tiene prioridad solo si shuffle estÃ¡ desactivado (BUG-16)
    if (dir === 1 && localQueue.length > 0 && !shuffle) {
      const next = localQueue[0];
      setLocalQueue((q) => q.slice(1));
      setActiveTrack(next);
      setAutoPlaySignal((n) => n + 1);
      return;
    }
    const list = filteredTracks ?? tracks;
    if (!list.length) return;
    const idx = list.findIndex((t) => t.id === activeTrack?.id);
    const base = idx < 0 ? 0 : idx;
    // BUG-09: loopMode "off" â†’ parar al llegar al Ãºltimo track
    if (dir === 1 && loopModeRef.current === "off" && !shuffle && base >= list.length - 1) {
      setIsPlaying(false);
      return;
    }
    const nextIdx = shuffle
      ? Math.floor(Math.random() * list.length)
      : (base + dir + list.length) % list.length;
    setActiveTrack(list[nextIdx]);
    setAutoPlaySignal((n) => n + 1);
  }, [tracks, localQueue, activeTrack?.id, shuffle]);

  // refs para que callbacks de YT no tengan stale closures
  const skipRef     = useRef(null);
  const loopModeRef = useRef("off");
  useEffect(() => { skipRef.current     = skip;     }, [skip]);
  useEffect(() => { loopModeRef.current = loopMode; }, [loopMode]);

  // â”€â”€ Handlers cola â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const addToQueue = useCallback((track) => {
    if (!track) return;
    setLocalQueue((q) => q.find((t) => t.id === track.id) ? q : [...q, track]);
  }, []);

  const removeFromQueue = useCallback((trackId) => {
    setLocalQueue((q) => {
      const isActive = activeTrack?.id === trackId;
      const filtered = q.filter((t) => t.id !== trackId);
      if (isActive) {
        const next = filtered.find((t) => t.id !== trackId);
        if (next) { setActiveTrack(next); setAutoPlaySignal((n) => n + 1); }
        else setActiveTrack(tracks[0] ?? null);
      }
      return filtered;
    });
  }, [activeTrack?.id, tracks]);

  const playTrack = useCallback((track) => {
    if (activeTrack?.id === track.id) {
      // BUG-11: tap en track activo â†’ toggle play/pause
      if (isPlaying) {
        if (isYouTube) { ytPlayerRef.current?.pauseVideo?.(); }
        else           { audioRef.current?.pause(); }
        setIsPlaying(false);
      } else {
        if (isYouTube) { ytPlayerRef.current?.playVideo?.(); }
        else           { audioRef.current?.play().catch(() => {}); }
        setIsPlaying(true);
      }
      return;
    }
    setActiveTrack(track);
    setAutoPlaySignal((n) => n + 1);
  }, [activeTrack?.id, isPlaying, isYouTube]);

  const toggleLoopMode = useCallback(() => {
    setLoopMode((m) => m === "off" ? "list" : m === "list" ? "track" : "off");
  }, []);

  // â”€â”€ WakeLock â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    if (!("wakeLock" in navigator)) return;
    const req = async () => {
      try { wakeLockRef.current = await navigator.wakeLock.request("screen"); } catch (_) {}
    };
    const rel = async () => {
      if (wakeLockRef.current) { await wakeLockRef.current.release(); wakeLockRef.current = null; }
    };
    const onVis = () => { if (document.visibilityState === "visible" && isPlaying) req(); };
    isPlaying ? req() : rel();
    document.addEventListener("visibilitychange", onVis);
    return () => { rel(); document.removeEventListener("visibilitychange", onVis); };
  }, [isPlaying]);

  // â”€â”€ Vibe accent â€” CSS variable global â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    const color = VIBE_COLORS[activeVibe] ?? null;
    if (color) {
      document.documentElement.style.setProperty("--vibe-accent", color);
    } else {
      document.documentElement.style.removeProperty("--vibe-accent");
    }
  }, [activeVibe]);

   // â”€â”€ InicializaciÃ³n YT.Player persistente â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    if (!activeTrack) return;
    const isYT = activeTrack.type === "youtube" || (!activeTrack.type && activeTrack.url);
    if (!isYT) return;

    const initPlayer = () => {
      const videoId = extractYouTubeId(activeTrack.youtube_url ?? activeTrack.url ?? "");
      // Si el player ya existe, solo cambiar video â€” nunca destroy() sobre div controlado por React
      if (ytPlayerRef.current && typeof ytPlayerRef.current.loadVideoById === "function") {
        ytPlayerRef.current.loadVideoById(videoId);
        return;
      }
      ytPlayerRef.current = new window.YT.Player("yt-player-div", {
        videoId,
        playerVars: { autoplay: 0, controls: 0, rel: 0 },
        events: {
          onReady: (e) => {
            e.target.setVolume(masterVolume ?? 80);
          },
          onStateChange: (e) => {
            const s = e.data;
            if (s === window.YT.PlayerState.PLAYING)     { setIsPlaying(true); }
            else if (s === window.YT.PlayerState.PAUSED) { setIsPlaying(false); }
            else if (s === window.YT.PlayerState.ENDED)  {
              if (loopModeRef.current === "track") {
                e.target.seekTo(0); e.target.playVideo();
              } else {
                setIsPlaying(false); skipRef.current?.(1);
              }
            }
          },
        },
      });
    };

    loadYTApi(initPlayer);
    // NO cleanup destroy aquÃ­ â€” el player debe sobrevivir cambios de pestaÃ±a
  }, [activeTrack?.id]);

  // â”€â”€ onEnded centralizado â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const handleAudioEnded = useCallback(() => {
    if (loopMode === "track" && audioRef.current) {
      audioRef.current.currentTime = 0;
      audioRef.current.play().catch(() => {});
    } else {
      skip(1);
    }
  }, [loopMode, skip]);

  // â”€â”€ src del audio: reactivo al track activo â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const audioSrc = (!activeTrack || activeTrack.type === "youtube")
    ? null
    : activeTrack.drive_file_id
      ? getStreamUrl(activeTrack.drive_file_id)
      : null;
  // â”€â”€ div YT fuera del Ã¡rbol React â€” imperativo, nunca en JSX â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    if (document.getElementById("yt-player-div")) return;
    const div = document.createElement("div");
    div.id = "yt-player-div";
    div.style.display = "none";
    document.body.appendChild(div);
  }, []);

  return (
    <PlayerContext.Provider value={{
      activeTrack, setActiveTrack,
      isPlaying,   setIsPlaying,
      localQueue,  setLocalQueue,
      loopMode,    setLoopMode,    toggleLoopMode,
      shuffle,     setShuffle,
      autoPlaySignal, setAutoPlaySignal,
      syncStatus,
      activeVibe,   setActiveVibe,
      masterVolume, setMasterVolume,
      audioRef, ytPlayerRef,
      isYouTube,
      skip, addToQueue, removeFromQueue, playTrack,
    }}>
      {/* <audio> persistente â€” nunca se desmonta al cambiar de pestaÃ±a (ADR-004) */}
      <audio
        ref={audioRef}
        src={audioSrc ?? undefined}
        loop={false}
        style={{ display: "none" }}
        onEnded={handleAudioEnded}
        onTimeUpdate={(e) => {
          // WinampPlayer se suscribe a audioRef directamente para el progreso visual
        }}
      />
      {children}
    </PlayerContext.Provider>
  );
}

// â”€â”€â”€ BOOT PHASES â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
const BOOT = Object.freeze({
  IDLE:          "idle",
  INIT_SERVICES: "init_services",
  INIT_DB:       "init_db",
  READY:         "ready",
  UNAUTHORIZED:  "unauthorized",  // â† NUEVO: email no en la lista blanca
  ERROR:         "error",
});

// â”€â”€â”€ PROVIDER â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
export function VaultProvider({ children }) {

  // â”€â”€ State â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const [bootPhase,     setBootPhase]     = useState(BOOT.IDLE);
  const [user,          setUser]          = useState(null);   // { name, email, avatar, role, id }
  const [db,            setDb]            = useState(null);
  const [dbStatus,      setDbStatus]      = useState("OFFLINE");
  const [lastHeartbeat, setLastHeartbeat] = useState(null);
  const [activeTab,     setActiveTab]     = useState("vault");
  const [masterVolume,  setMasterVolume]  = useState(80);
  const [bootError,     setBootError]     = useState(null);
  const [pendingDrop,   setPendingDrop]   = useState(null);

  // Cache de capacidad por biblioteca { [libraryId]: { used_gb, used_pct, near_full } }
  const [libraryCapacity, setLibraryCapacity] = useState({});

  const hbTimerRef = useRef(null);

  // â”€â”€ Derived: authStatus â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const authStatus =
    bootPhase === BOOT.IDLE          ? "idle"          :
    bootPhase === BOOT.INIT_SERVICES ? "loading"       :
    bootPhase === BOOT.INIT_DB       ? "loading"       :
    bootPhase === BOOT.READY         ? "authenticated" :
    bootPhase === BOOT.UNAUTHORIZED  ? "unauthorized"  :
    "error";

  const isLoadingApp =
    bootPhase === BOOT.INIT_SERVICES ||
    bootPhase === BOOT.INIT_DB;

  // â”€â”€ Derived: bibliotecas con status recalculado â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  /**
   * Recalcula el status de cada biblioteca en cada render
   * basÃ¡ndose Ãºnicamente en fechas (sin red).
   */
  const libraries = useMemo(() => {
    if (!db?.libraries) return [];
    return db.libraries.map((lib) => ({
      ...lib,
      status: _calcLibraryStatus(lib.last_heartbeat_check),
    }));
  }, [db?.libraries]);

  /** Bibliotecas en riesgo de inactividad (18-22 meses sin heartbeat). */
  const librariesAtRisk = useMemo(
    () => libraries.filter((l) => l.status === "at_risk"),
    [libraries]
  );

  /** Bibliotecas presumiblemente cerradas por Google (â‰¥23 meses). */
  const librariesLost = useMemo(
    () => libraries.filter((l) => l.status === "lost"),
    [libraries]
  );

  /**
   * Bibliotecas con capacidad casi agotada.
   * Solo las que tengamos en el cache de capacidad.
   */
  const librariesNearFull = useMemo(
    () => libraries.filter((l) => libraryCapacity[l.id]?.near_full === true),
    [libraries, libraryCapacity]
  );

  // â”€â”€ Derived: entradas filtradas por usuario actual â”€â”€â”€â”€â”€â”€â”€â”€
  /**
   * myEntries â€” todas las entradas dirigidas al usuario actual
   * (to === userId || to === "both")
   */
  const myEntries = useMemo(() => {
    if (!db?.entries || !user) return [];
    return db.entries.filter(
      (e) => e.to === "both" || e.to === user.id
    );
  }, [db?.entries, user?.id]);

  /**
   * myMessages â€” solo type "letter" dirigidas al usuario actual
   * (cartas de Edu a David y viceversa)
   */
  const myMessages = useMemo(
    () => myEntries.filter((e) => e.type === "letter"),
    [myEntries]
  );

  // â”€â”€ Step 1: Init Google Services on mount â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  useEffect(() => {
    let cancelled = false;

    const boot = async () => {
      setBootPhase(BOOT.INIT_SERVICES);
      try {
        await initGoogleServices();
        if (cancelled) return;

        onAuthChange(({ event, token, error }) => {
          if (event === "SIGNED_IN"  && !cancelled) _onSignedIn(token);
          if (event === "SIGNED_OUT" && !cancelled) _onSignedOut();
          if (event === "AUTH_ERROR" && !cancelled) {
            console.error("[BÃºnker] Auth error:", error);
            setBootPhase(BOOT.ERROR);
            setBootError(error?.error_description ?? "Error de autenticaciÃ³n");
          }
        });

        if (isAuthenticated()) {
          await _fetchUserAndDb();
        } else if (restoreToken()) {
          await _fetchUserAndDb();
        } else {
          setBootPhase(BOOT.IDLE);
        }
      } catch (err) {
        if (!cancelled) {
          console.error("[BÃºnker] Boot failed:", err);
          setBootPhase(BOOT.ERROR);
          setBootError(err.message ?? "No se pudieron cargar los servicios de Google.");
        }
      }
    };

    boot();
    return () => { cancelled = true; };
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // â”€â”€ Step 2: Fetch user info + DB â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const _fetchUserAndDb = useCallback(async (token) => {
    setBootPhase(BOOT.INIT_DB);
    setDbStatus("SYNCING");

    try {
      const profile = await _fetchGoogleProfile(token);

      // â”€â”€ Control de acceso â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
      // Solo los tres emails autorizados pueden continuar.
      if (!isAuthorizedEmail(profile.email)) {
        console.warn(`[BÃºnker] Acceso denegado para: ${profile.email}`);
        // Cerramos la sesiÃ³n inmediatamente para no dejar token activo
        await signOut();
        setBootPhase(BOOT.UNAUTHORIZED);
        setBootError(
          `El email ${profile.email} no tiene acceso al BÃºnker.`
        );
        return;
      }

      // Enriquecemos el perfil con el rol interno
      const bunkerProfile = resolveUser(profile.email);
      const fullUser = {
        ...profile,
        role: bunkerProfile.role,
        id:   bunkerProfile.id,
        name: bunkerProfile.name, // usamos el nombre del BÃºnker, no el de Google
      };
      setUser(fullUser);

      const data = await initDatabase();
      setDb(data);
      setLastHeartbeat(data.heartbeat ?? new Date().toISOString());
      setDbStatus("ONLINE");
      setBootPhase(BOOT.READY);

      _scheduleHeartbeat();
    } catch (err) {
      console.error("[BÃºnker] DB init failed:", err);
      setDbStatus("ERROR");
      setBootPhase(BOOT.ERROR);
      setBootError(err.message ?? "Error al inicializar la base de datos.");
    }
  }, []);

  // â”€â”€ Auth event handlers â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const _onSignedIn = useCallback(async (token) => {
    await _fetchUserAndDb(token);
  }, [_fetchUserAndDb]);

  const _onSignedOut = useCallback(() => {
    setUser(null);
    setDb(null);
    setDbStatus("OFFLINE");
    setLastHeartbeat(null);
    setLibraryCapacity({});
    setBootPhase(BOOT.IDLE);
    _clearHeartbeat();
  }, []);

  // â”€â”€ Google profile helper â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const _fetchGoogleProfile = async (accessToken) => {
    try {
      const token = accessToken ?? window.gapi?.client?.getToken?.()?.access_token;
      if (!token) throw new Error("No hay token disponible");

      const res = await fetch(
        "https://www.googleapis.com/drive/v3/about?fields=user",
        { headers: { Authorization: `Bearer ${token}` } }
      );
      if (!res.ok) throw new Error(`Drive about failed: ${res.status}`);
      const json = await res.json();
      return {
        name:   json.user?.displayName ?? "Usuario",
        email:  json.user?.emailAddress ?? "",
        avatar: json.user?.photoLink    ?? null,
      };
    } catch (err) {
      throw new Error("[BÃºnker] No se pudo obtener el perfil: " + (err?.message ?? err));
    }
  };

  // â”€â”€ Heartbeat scheduler â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const _scheduleHeartbeat = useCallback(() => {
    _clearHeartbeat();
    hbTimerRef.current = setInterval(async () => {
      try {
        await triggerHeartbeat();
        setLastHeartbeat(new Date().toISOString());
      } catch (e) {
        console.warn("[BÃºnker] Scheduled heartbeat failed:", e);
      }
    }, 20 * 60 * 1000);
  }, []);

  const _clearHeartbeat = useCallback(() => {
    if (hbTimerRef.current) {
      clearInterval(hbTimerRef.current);
      hbTimerRef.current = null;
    }
  }, []);

  useEffect(() => () => _clearHeartbeat(), [_clearHeartbeat]);

  // â”€â”€ Drop polling global (ERROR 33) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  // Detecta drops entrantes independientemente de la ruta activa.
  useEffect(() => {
    if (!user || !db) return;
    const checkDrops = () => {
      const drops = db?.music_lab?.drops ?? [];
      const unseen = drops.find((d) => !d.seen && d.to === user.id);
      if (unseen) setPendingDrop(unseen);
    };
    checkDrops();
    const interval = setInterval(async () => {
      try {
        const fresh = await fetchData();
        const unseen = (fresh?.music_lab?.drops ?? []).find(
          (d) => !d.seen && d.to === user.id
        );
        if (unseen) setPendingDrop(unseen);
      } catch (e) {
        console.warn("[BÃºnker] Drop polling fallÃ³ silenciosamente:", e);
      }
    }, 30 * 1000);
    return () => clearInterval(interval);
  }, [user?.id, db?.music_lab?.drops]);

  // â”€â”€ PUBLIC ACTIONS: Auth â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  const login = useCallback(async () => {
    setBootPhase(BOOT.INIT_SERVICES);
    try {
      await requestAccessToken(true);
      // _onSignedIn se dispara vÃ­a onAuthChange
    } catch (err) {
      console.error("[BÃºnker] Login error:", err);
      setBootPhase(BOOT.ERROR);
      setBootError(err.message);
    }
  }, []);

  const logout = useCallback(async () => {
    try {
      await signOut();
    } catch (e) {
      console.warn("[BÃºnker] Sign out error:", e);
    }
    _onSignedOut();
  }, [_onSignedOut]);

  const refreshDb = useCallback(async () => {
    if (!isAuthenticated()) return;
    setDbStatus("SYNCING");
    try {
      const fresh = await fetchData();
      setDb(fresh);
      setLastHeartbeat(fresh.heartbeat ?? new Date().toISOString());
      setDbStatus("ONLINE");
    } catch (err) {
      console.error("[BÃºnker] refreshDb failed:", err);
      setDbStatus("ERROR");
    }
  }, []);

  /**
   * patchDb(updater)
   * ActualizaciÃ³n optimista local + persistencia en Drive.
   * updater = (currentDb) => newDb
   */
  const patchDb = useCallback(async (updater) => {
    if (!isAuthenticated()) return;
    setDbStatus("SYNCING");
    try {
      setDb((prev) => updater(prev));
      await updateData(updater);
      setDbStatus("ONLINE");
    } catch (err) {
      console.error("[BÃºnker] patchDb failed:", err);
      setDbStatus("ERROR");
      await refreshDb();
    }
  }, [refreshDb]);

  const heartbeat = useCallback(async () => {
    try {
      await triggerHeartbeat();
      setLastHeartbeat(new Date().toISOString());
    } catch (e) {
      console.warn("[BÃºnker] Manual heartbeat failed:", e);
    }
  }, []);

  // â”€â”€ Page Visibility: heartbeat al volver de background â”€â”€â”€â”€
  useEffect(() => {
    const onVisible = () => {
      if (document.visibilityState === "visible" && bootPhase === BOOT.READY) {
        heartbeat();
      }
    };
    document.addEventListener("visibilitychange", onVisible);
    return () => document.removeEventListener("visibilitychange", onVisible);
  }, [bootPhase, heartbeat]);

  // â”€â”€ PUBLIC ACTIONS: Bibliotecas â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  /**
   * addLibrary(data)
   * AÃ±ade una biblioteca y actualiza el estado local optimistamente.
   */
  const addLibrary = useCallback(async (data) => {
    const newLib = await svcAddLibrary(data);
    setDb((prev) => ({
      ...prev,
      libraries: [...(prev?.libraries ?? []), newLib],
    }));
    return newLib;
  }, []);

  /**
   * updateLibrary(id, patch)
   * Actualiza una biblioteca y sincroniza el estado local.
   */
  const updateLibrary = useCallback(async (id, patch) => {
    const updated = await svcUpdateLibrary(id, patch);
    setDb((prev) => ({
      ...prev,
      libraries: (prev?.libraries ?? []).map((l) => (l.id === id ? updated : l)),
    }));
    return updated;
  }, []);

  /**
   * removeLibrary(id)
   * Elimina una biblioteca y limpia su cache de capacidad.
   */
  const removeLibrary = useCallback(async (id) => {
    await svcRemoveLibrary(id);
    setDb((prev) => ({
      ...prev,
      libraries: (prev?.libraries ?? []).filter((l) => l.id !== id),
    }));
    setLibraryCapacity((prev) => {
      const next = { ...prev };
      delete next[id];
      return next;
    });
  }, []);

  /**
   * markLibraryHeartbeat(id)
   * El admin confirma que acaba de entrar a la cuenta biblioteca.
   */
  const markLibraryHeartbeat = useCallback(async (id) => {
    const updated = await svcMarkLibraryHeartbeat(id);
    setDb((prev) => ({
      ...prev,
      libraries: (prev?.libraries ?? []).map((l) => (l.id === id ? updated : l)),
    }));
    return updated;
  }, []);

  /**
   * getLibraryContents(sharedFolderId)
   * Lista los archivos de una carpeta compartida.
   */
  const getLibraryContents = useCallback(
    (sharedFolderId) => listLibraryContents(sharedFolderId),
    []
  );

  /**
   * refreshLibraryCapacity(library)
   * Calcula el uso actual de una biblioteca y actualiza el cache.
   * Llama a Drive API â€” Ãºsalo con moderaciÃ³n.
   *
   * @param {{ id, shared_folder_id, capacity_gb }} library
   * @returns {Promise<{ used_gb, used_pct, near_full, file_count }>}
   */
  const refreshLibraryCapacity = useCallback(async (library) => {
    const info = await checkLibraryCapacity(library);
    setLibraryCapacity((prev) => ({ ...prev, [library.id]: info }));

    // Si la biblioteca estÃ¡ casi llena, actualizamos su status a "full" en la DB
    if (info.used_pct >= 100) {
      await svcUpdateLibrary(library.id, { status: "full" });
      setDb((prev) => ({
        ...prev,
        libraries: (prev?.libraries ?? []).map((l) =>
          l.id === library.id ? { ...l, status: "full" } : l
        ),
      }));
    }

    return info;
  }, []);

  // â”€â”€ Context Value â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const value = {
    // Auth
    user,
    isLoading: isLoadingApp,
    authStatus,
    bootPhase,
    bootError,

    // Database
    db,
    dbStatus,
    lastHeartbeat,

    // NavegaciÃ³n
    activeTab,
    setActiveTab,

    // MÃºsica
    masterVolume,
    setMasterVolume,

    // Entradas filtradas por usuario
    myEntries,
    myMessages,

    // Bibliotecas
    libraries,
    librariesAtRisk,
    librariesLost,
    librariesNearFull,
    libraryCapacity,

    // Acciones core
    login,
    logout,
    refreshDb,
    patchDb,
    heartbeat,

    // Acciones de biblioteca
    addLibrary,
    updateLibrary,
    removeLibrary,
    markLibraryHeartbeat,
    getLibraryContents,
    refreshLibraryCapacity,

    // Drops globales (ERROR 33)
    pendingDrop,
    clearPendingDrop: (dropId) => {
      setPendingDrop(null);
      if (dropId) {
        patchDb((db) => ({
          ...db,
          music_lab: {
            ...db.music_lab,
            drops: (db.music_lab?.drops ?? []).map((d) =>
              d.id === dropId ? { ...d, seen: true } : d
            ),
          },
        }));
      }
    },

    // Servicio raw para usos avanzados
    driveService: GoogleDriveService,
  };

  return (
    <VaultContext.Provider value={value}>
      {children}
    </VaultContext.Provider>
  );
}

export function useVault() {
  const ctx = useContext(VaultContext);
  if (!ctx) throw new Error("[useVault] Must be used inside <VaultProvider>");
  return ctx;
}

const MusicPage = () => {
  const { db, patchDb, user, masterVolume, setMasterVolume } = useVault();
  return (
    <MusicLab
      db={db}
      userId={user?.id ?? null}
      onAddTrack={async (trackData) => {
        await patchDb((prev) => ({
          ...prev,
          music_lab: {
            ...prev.music_lab,
            tracks: [...(prev.music_lab?.tracks ?? []), trackData],
          },
        }));
      }}
      onAddDrop={async (dropData) => {
        await patchDb((prev) => ({
          ...prev,
          music_lab: {
            ...prev.music_lab,
            drops: [...(prev.music_lab?.drops ?? []), dropData],
          },
        }));
      }}
      onDeleteTrack={async (trackId) => {
        await patchDb((prev) => ({
          ...prev,
          music_lab: {
            ...prev.music_lab,
            tracks: (prev.music_lab?.tracks ?? []).filter((t) => t.id !== trackId),
          },
        }));
      }}
      onEditTrack={async (updatedTrack) => {
        await patchDb((prev) => ({
          ...prev,
          music_lab: {
            ...prev.music_lab,
            tracks: (prev.music_lab?.tracks ?? []).map((t) =>
              t.id === updatedTrack.id ? { ...t, ...updatedTrack } : t
            ),
          },
        }));
      }}
      onMasterVolumeChange={setMasterVolume}
      onMarkDropSeen={async (dropId) => {
        await patchDb((prev) => ({
          ...prev,
          music_lab: {
            ...prev.music_lab,
            drops: (prev.music_lab?.drops ?? []).map((d) =>
              d.id === dropId ? { ...d, seen: true } : d
            ),
          },
        }));
      }}
    />
  );
};

const useGlitch = (text, interval = 80, glitchDuration = 600) => {
  const [display, setDisplay] = useState(text);
  const timerRef = useRef(null);

  useEffect(() => {
    let elapsed = 0;
    const glitch = () => {
      if (elapsed >= glitchDuration) {
        setDisplay(text);
        return;
      }
      setDisplay(
        text
          .split("")
          .map((char, i) =>
            char !== " " && Math.random() < 0.25
              ? GLITCH_CHARS[Math.floor(Math.random() * GLITCH_CHARS.length)]
              : char
          )
          .join("")
      );
      elapsed += interval;
      timerRef.current = setTimeout(glitch, interval);
    };
    timerRef.current = setTimeout(glitch, 600);
    return () => clearTimeout(timerRef.current);
  }, [text, interval, glitchDuration]);

  return display;
};

import { useRef, useEffect, useCallback } from "react";
import { ACID } from "../layout/Layout";

export const AudioVisualizer = ({ audioRef, isPlaying, accentColor = ACID }) => {
  const canvasRef    = useRef(null);
  const analyserRef  = useRef(null);
  const sourceRef    = useRef(null);
  const animFrameRef = useRef(null);
  const audioCtxRef  = useRef(null);

  const setupAudio = useCallback(() => {
    if (!audioRef?.current || audioCtxRef.current) return;
    try {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      if (!AudioContext) return;
      const ctx      = new AudioContext();
      const analyser = ctx.createAnalyser();
      analyser.fftSize = 128;
      analyser.smoothingTimeConstant = 0.82;
      const source = ctx.createMediaElementSource(audioRef.current);
      source.connect(analyser);
      analyser.connect(ctx.destination);
      audioCtxRef.current  = ctx;
      analyserRef.current  = analyser;
      sourceRef.current    = source;
    } catch (e) {}
  }, [audioRef]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx    = canvas.getContext("2d");
    const BAR_COUNT = 32;

    const draw = () => {
      animFrameRef.current = requestAnimationFrame(draw);
      const W = canvas.width;
      const H = canvas.height;
      if (!W || !H) return;
      ctx.clearRect(0, 0, W, H);

      let dataArray;
      if (analyserRef.current && isPlaying) {
        const bufferLen = analyserRef.current.frequencyBinCount;
        dataArray = new Uint8Array(bufferLen);
        analyserRef.current.getByteFrequencyData(dataArray);
      }

      const barW = (W / BAR_COUNT) - 1.5;
      const now  = performance.now() / 1000;

      for (let i = 0; i < BAR_COUNT; i++) {
        let barH;
        if (dataArray && isPlaying) {
          const idx = Math.floor((i / BAR_COUNT) * dataArray.length);
          barH = (dataArray[idx] / 255) * H * 0.9;
        } else {
          const base   = Math.sin(now * 1.2 + i * 0.35) * 0.38 + 0.52;
          const ripple = Math.sin(now * 3.1 + i * 0.7)  * 0.12;
          barH = Math.max((base + ripple) * H * 0.72, H * 0.08);
        }

        const x    = i * (barW + 1.5);
        const y    = H - barH;
        const grad = ctx.createLinearGradient(x, y, x, H);
        const hex  = accentColor;
        grad.addColorStop(0,   hex + "ff");
        grad.addColorStop(0.5, hex + "aa");
        grad.addColorStop(1,   hex + "22");

        ctx.fillStyle   = grad;
        ctx.shadowBlur  = barH > H * 0.5 ? 18 : 6;
        ctx.shadowColor = hex + "88";

        const radius = Math.min(3, barW / 2);
        ctx.beginPath();
        ctx.moveTo(x + radius, y);
        ctx.lineTo(x + barW - radius, y);
        ctx.quadraticCurveTo(x + barW, y, x + barW, y + radius);
        ctx.lineTo(x + barW, H);
        ctx.lineTo(x, H);
        ctx.lineTo(x, y + radius);
        ctx.quadraticCurveTo(x, y, x + radius, y);
        ctx.fill();

        if (isPlaying && barH > H * 0.3) {
          ctx.fillStyle   = "#ffffff88";
          ctx.shadowBlur  = 12;
          ctx.shadowColor = hex;
          ctx.fillRect(x, y - 3, barW, 2);
        }
        ctx.shadowBlur = 0;
      }
    };

    let drawStarted = false;
    const tryStart = () => {
      const w = canvas.offsetWidth;
      const h = canvas.offsetHeight;
      if (w === 0 || h === 0) {
        requestAnimationFrame(tryStart);
        return;
      }
      canvas.width  = w * window.devicePixelRatio;
      canvas.height = h * window.devicePixelRatio;
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
      if (!drawStarted) { drawStarted = true; draw(); }
    };
    const resize = () => {
      const w = canvas.offsetWidth;
      const h = canvas.offsetHeight;
      if (w === 0 || h === 0) return;
      canvas.width  = w * window.devicePixelRatio;
      canvas.height = h * window.devicePixelRatio;
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
    };
    tryStart();
    window.addEventListener("resize", resize);
    return () => {
      cancelAnimationFrame(animFrameRef.current);
      window.removeEventListener("resize", resize);
    };
  }, [isPlaying, accentColor]);

  useEffect(() => {
    if (isPlaying) {
      setupAudio();
      audioCtxRef.current?.resume();
    }
  }, [isPlaying, setupAudio]);

  return (
    <canvas
      ref={canvasRef}
      className="w-full"
      style={{ height: 72, display: "block" }}
    />
  );
};

const CircularRing = ({ progress, color, radius = 90, stroke = 5, children }) => {

          cx={size / 2}
          cy={size / 2}
          r={radius}
          fill="none"

        />

        <circle
          cx={size / 2}
          cy={size / 2}
          r={radius}

/** Returns { years, months, days } since a given ISO date string */
const timeSince = (isoDate) => {
  if (!isoDate) return { years: 0, months: 0, days: 0, totalDays: 0 };
  const start = new Date(isoDate);
  const now   = new Date();
  let years   = now.getFullYear() - start.getFullYear();
  let months  = now.getMonth()    - start.getMonth();
  let days    = now.getDate()     - start.getDate();
  if (days   < 0) { months--; days   += new Date(now.getFullYear(), now.getMonth(), 0).getDate(); }
  if (months < 0) { years--;  months += 12; }
  const totalDays = Math.floor((now - start) / 86400000);
  return { years, months, days, totalDays };
};

/** Returns time remaining until a future ISO date */
const timeUntil = (isoDate) => {
  if (!isoDate) return null;
  const target = new Date(isoDate);
  const now    = new Date();
  const diff   = target - now;
  if (diff <= 0) return { unlocked: true };
  const days    = Math.floor(diff / 86400000);
  const hours   = Math.floor((diff % 86400000) / 3600000);
  const minutes = Math.floor((diff % 3600000) / 60000);
  return { unlocked: false, days, hours, minutes };
};

/** Months since a given ISO date */
const monthsSince = (isoDate) => {
  if (!isoDate) return 0;
  const start = new Date(isoDate);
  const now   = new Date();
  return (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
};
